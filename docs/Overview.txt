package checker

import (
	"net/http"
	"sync"
	"time"
)

type CheckResult struct {
	URL        string        `json:"url"`
	StatusCode int           `json:"status_code"`
	Latency    time.Duration `json:"latency"`
	Error      string        `json:"error,omitempty"`
}

// RunPool distributes target URLs across a set number of concurrent workers
func RunPool(urls []string, workerCount int, timeout time.Duration) []CheckResult {
	jobs := make(chan string, len(urls))
	resultsChan := make(chan CheckResult, len(urls))
	var wg sync.WaitGroup

	client := &http.Client{Timeout: timeout}

	// Spin up workers
	for w := 1; w <= workerCount; w++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for url := range jobs {
				start := time.Now()
				resp, err := client.Get(url)
				latency := time.Since(start)

				if err != nil {
					resultsChan <- CheckResult{URL: url, Latency: latency, Error: err.Error()}
					continue
				}
				resp.Body.Close()
				resultsChan <- CheckResult{URL: url, StatusCode: resp.StatusCode, Latency: latency}
			}
		}()
	}

	// Feed jobs
	for _, url := range urls {
		jobs <- url
	}
	close(jobs)

	// Wait and collect
	wg.Wait()
	close(resultsChan)

	var results []CheckResult
	for res := range resultsChan {
		results = append(results, res)
	}
	return results
}
